    });

    return NextResponse.json({ success: true, message: 'Lead oluşturuldu ve Telegram bildirimi gönderildi.' });
  } catch (error) {
    return NextResponse.json({ error: 'Sunucu hatası' }, { status: 500 });
  }
}`}
              </pre>
            </div>
          </div>
        )}
      </div>

      {/* --- LEAD DETAY MODAL --- */}
      {selectedLead && (
        <div className="fixed inset-0 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-4 z-50">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-lg w-full p-6 shadow-2xl relative">
            <button
              onClick={() => setSelectedLead(null)}
              className="absolute top-4 right-4 text-slate-400 hover:text-white text-lg font-bold"
            >
              ✕
            </button>

            <div className="flex items-center gap-2 text-xs text-slate-400 mb-2">
              <span>{SOURCE_CONFIG[selectedLead.source]?.icon}</span>
              <span>{SOURCE_CONFIG[selectedLead.source]?.label}</span>
              <span>•</span>
              <span>{new Date(selectedLead.createdAt).toLocaleString('tr-TR')}</span>
            </div>

            <h3 className="text-xl font-bold text-white">{selectedLead.fullName}</h3>
            <p className="text-xs text-slate-400 mb-4">{selectedLead.company}</p>

            <div className="space-y-4">
              {/* Durum Değiştirici */}
              <div>
                <label className="block text-[11px] text-slate-400 mb-1">Süreç Durumu</label>
                <select
                  value={selectedLead.status}
                  onChange={(e) => handleStatusChange(selectedLead.id, e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 text-slate-200 text-xs rounded-xl p-2.5 focus:outline-none focus:border-cyan-500 font-semibold"
                >
                  {Object.keys(STATUS_CONFIG).map((st) => (
                    <option key={st} value={st}>
                      {STATUS_CONFIG[st].label}
                    </option>
                  ))}
                </select>
              </div>

              {/* Hızlı İletişim Butonları */}
              <div className="grid grid-cols-2 gap-3">
                <a
                  href={`tel:${selectedLead.phone}`}
                  className="bg-slate-800 hover:bg-slate-700 text-slate-200 text-xs text-center py-2 rounded-xl transition-colors font-medium flex items-center justify-center gap-2"
                >
                  📞 Ara ({selectedLead.phone})
                </a>
                <a
                  href={`https://wa.me/${selectedLead.phone?.replace(/[^0-0]/g, '')}`}
                  target="_blank"
                  rel="noreferrer"
                  className="bg-emerald-600/20 hover:bg-emerald-600/30 text-emerald-400 border border-emerald-500/30 text-xs text-center py-2 rounded-xl transition-colors font-medium flex items-center justify-center gap-2"
                >
                  💬 WhatsApp Mesajı
                </a>
              </div>

              {/* Not Alanı */}
              <div>
                <label className="block text-[11px] text-slate-400 mb-1">Görüşme Notları / Takip Bilgisi</label>
                <textarea
                  rows={4}
                  value={selectedLead.notes || ''}
                  onChange={(e) => handleNoteUpdate(selectedLead.id, e.target.value)}
                  placeholder="Müşteriyle yapılan görüşme detaylarını buraya yazın..."
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl p-3 text-xs text-slate-200 placeholder-slate-600 focus:outline-none focus:border-cyan-500"
                />
              </div>
            </div>

            <div className="mt-6 pt-4 border-t border-slate-800 flex justify-end">
              <button
                onClick={() => setSelectedLead(null)}
                className="bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-bold text-xs px-5 py-2 rounded-xl transition-all"
              >
                Kaydet ve Kapat
              </button>
            </div>
          </div>
        </div>
      )}

      {/* --- MANUEL LEAD EKLE MODAL --- */}
      {isNewLeadModalOpen && (
        <div className="fixed inset-0 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-4 z-50">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-lg w-full p-6 shadow-2xl relative">
            <button
              onClick={() => setIsNewLeadModalOpen(false)}
              className="absolute top-4 right-4 text-slate-400 hover:text-white text-lg font-bold"
            >
              ✕
            </button>

            <h3 className="text-lg font-bold text-white mb-4">Manuel Potansiyel Müşteri Ekle</h3>

            <form onSubmit={handleCreateLead} className="space-y-3 text-xs">
              <div>
                <label className="block text-slate-400 mb-1">Ad Soyad *</label>
                <input
                  type="text"
                  required
                  value={newLeadForm.fullName}
                  onChange={(e) => setNewLeadForm({ ...newLeadForm, fullName: e.target.value })}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-200 focus:outline-none focus:border-cyan-500"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-slate-400 mb-1">Şirket Adı</label>
                  <input
                    type="text"
                    value={newLeadForm.company}
                    onChange={(e) => setNewLeadForm({ ...newLeadForm, company: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-200 focus:outline-none focus:border-cyan-500"
                  />
                </div>
                <div>
                  <label className="block text-slate-400 mb-1">Telefon</label>
                  <input
                    type="text"
                    value={newLeadForm.phone}
                    onChange={(e) => setNewLeadForm({ ...newLeadForm, phone: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-200 focus:outline-none focus:border-cyan-500"
                  />
                </div>
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-slate-400 mb-1">Hizmet</label>
                  <select
                    value={newLeadForm.service}
                    onChange={(e) => setNewLeadForm({ ...newLeadForm, service: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-200 focus:outline-none focus:border-cyan-500"
                  >
                    <option value="Sızma Testi (Pentest)">Sızma Testi (Pentest)</option>
                    <option value="ISO 27001 Danışmanlığı">ISO 27001 Danışmanlığı</option>
                    <option value="Özel Web Yazılımı">Özel Web Yazılımı</option>
                    <option value="Ücretsiz Güvenlik Analizi">Ücretsiz Güvenlik Analizi</option>
                  </select>
                </div>
                <div>
                  <label className="block text-slate-400 mb-1">Tahmini Değer (₺)</label>
                  <input
                    type="number"
                    value={newLeadForm.value}
                    onChange={(e) => setNewLeadForm({ ...newLeadForm, value: e.target.value })}
                    className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-200 focus:outline-none focus:border-cyan-500"
                  />
                </div>
              </div>

              <div className="mt-4 flex justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsNewLeadModalOpen(false)}
                  className="px-4 py-2 text-slate-400 hover:text-white"
                >
                  İptal
                </button>
                <button
                  type="submit"
                  className="bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-bold px-5 py-2 rounded-xl transition-all"
                >
                  Lead Oluştur
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
